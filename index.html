window.addEventListener('DOMContentLoaded', async () => { const savedKey = localStorage.getItem('ga_active_license'); const savedScriptUrl = localStorage.getItem('ga_script_url'); const savedCompanyName = localStorage.getItem('ga_company_name'); if (savedKey && savedScriptUrl) { // Page refresh korlei live server theke check korbe key active ache kina try { const res = await fetch(`${LICENSE_API_URL}?key=${encodeURIComponent(savedKey)}`); const data = await res.json(); if (data.status === "SUCCESS") { API_URL = data.scriptUrl; document.getElementById('licenseLoginModal').classList.add('hidden'); updatePortalHeader(data.company || savedCompanyName); // Days remaining calculate kora const expDate = new Date(data.expiry); const now = new Date(); now.setHours(0, 0, 0, 0); const daysLeft = Math.ceil((expDate - now) / (1000 * 60 * 60 * 24)); updateLicenseCounterDisplay(daysLeft, data.expiry); fetchSheetData(); return; } else { // Jodi sheet-e Block ba Expired kore deya hoy, tahole data clear kore login modal ante hobe alert("Access Denied: " + data.message); localStorage.clear(); } } catch (e) { // Internet na thakle ba server error hole purono localStorage diye chalabe (optional) API_URL = savedScriptUrl; document.getElementById('licenseLoginModal').classList.add('hidden'); updatePortalHeader(savedCompanyName); fetchSheetData(); return; } } // Jodi key na thake ba block hoye jay document.getElementById('licenseLoginModal').classList.remove('hidden'); });
